Atomic database transaction (all database writes in one transaction)

The following SQL (PostgreSQL syntax) encapsulates the entire atomic logic.
Assume application variables are passed in:
v_init_telegram_id, v_init_username, v_init_first_name,
v_contact_phone, v_referral_code (the two telegram IDs have already been verified).

sql
DO $$
DECLARE
    v_init_telegram_id   BIGINT := ...;   -- from verified initData
    v_init_username      TEXT   := ...;   -- from initData
    v_init_first_name    TEXT   := ...;   -- from initData
    v_contact_phone      VARCHAR(25) := ...; -- from verified contact (already normalized, e.g., +251912345678)
    v_referral_code      BIGINT := ...;   -- from localStorage, NULL if none

    referrer             RECORD;
    reg_bonus            NUMERIC(12,2);
    referral_amt         NUMERIC(12,2);
    referral_limit       INT;
BEGIN
    -- Fetch current game_config values (single row, id=1)
    SELECT registration_bonus, referral_amount, referral_monthly_limit
    INTO reg_bonus, referral_amt, referral_limit
    FROM game_config
    WHERE id = 1;

    -- Idempotency: check if user already exists by telegram_id
    PERFORM 1 FROM users WHERE telegram_id = v_init_telegram_id;
    IF FOUND THEN
        -- User already registered; no changes needed.
        -- The application will simply return the existing user.
    ELSE
        -- Check if phone_number already exists (different telegram_id)
        PERFORM 1 FROM users WHERE phone_number = v_contact_phone;
        IF FOUND THEN
            -- Update existing user with new telegram_id
            UPDATE users
            SET telegram_id = v_init_telegram_id,
                username = COALESCE(v_init_username, 'Player'),
                first_name = COALESCE(LEFT(v_init_first_name, 12), 'Player')
            WHERE phone_number = v_contact_phone;
        ELSE
            -- Create new user
            INSERT INTO users (
                telegram_id,
                phone_number,
                username,
                first_name,
                non_withdrawal_wallet,
                referral_count,
                last_referred_date,
                created_at
            ) VALUES (
                v_init_telegram_id,
                v_contact_phone,
                COALESCE(v_init_username, 'Player'),
                COALESCE(LEFT(v_init_first_name, 12), 'Player'),
                reg_bonus,
                0,
                NULL,
                NOW()
            );

            -- Handle referral if provided and not self-referral
            IF v_referral_code IS NOT NULL AND v_referral_code != v_init_telegram_id THEN
                -- Lock the referrer row to prevent race conditions
                SELECT * INTO referrer
                FROM users
                WHERE telegram_id = v_referral_code
                FOR UPDATE;

                IF FOUND THEN
                    -- Reset referral count if month changed or first referral
                    IF referrer.last_referred_date IS NULL
                       OR EXTRACT(MONTH FROM referrer.last_referred_date) != EXTRACT(MONTH FROM NOW()) THEN
                        UPDATE users
                        SET referral_count = 0,
                            last_referred_date = NOW()
                        WHERE telegram_id = v_referral_code
                        RETURNING referral_count, last_referred_date
                        INTO referrer.referral_count, referrer.last_referred_date;
                    END IF;

                    -- Now referrer.referral_count reflects the current value after any reset
                    IF referrer.referral_count < referral_limit THEN
                        -- Credit referrer (add to non_withdrawal_wallet)
                        UPDATE users
                        SET non_withdrawal_wallet = non_withdrawal_wallet + referral_amt,
                            referral_count = referral_count + 1,
                            last_referred_date = NOW()
                        WHERE telegram_id = v_referral_code;

                        -- Insert rewarded referral history
                        INSERT INTO referrals_history (
                            referrer_id,
                            referred_user_id,
                            rewarded,
                            rewarded_amount,
                            created_at
                        ) VALUES (
                            v_referral_code,
                            v_init_telegram_id,
                            TRUE,
                            referral_amt,
                            NOW()
                        );
                    ELSE
                        -- Monthly limit exceeded: insert unrewarded history
                        INSERT INTO referrals_history (
                            referrer_id,
                            referred_user_id,
                            rewarded,
                            rewarded_amount,
                            created_at
                        ) VALUES (
                            v_referral_code,
                            v_init_telegram_id,
                            FALSE,
                            0,
                            NOW()
                        );
                    END IF;
                END IF;
                -- If referrer not found, ignore referral (no action)
            END IF;
        END IF;
    END IF;

    COMMIT;
END $$;
All writes are atomic. After the transaction commits, the registration is final.

CREATE OR REPLACE FUNCTION redeem_coupon(
    p_coupon_code  TEXT,
    p_telegram_id  BIGINT
) RETURNS JSONB
LANGUAGE plpgsql
AS $$
DECLARE
    v_coupon       coupons%ROWTYPE;
    v_claimed_id   BIGINT;
BEGIN
    -- 1. Lock the coupon row to prevent concurrent redemptions
    SELECT * INTO v_coupon
    FROM coupons
    WHERE coupon_code = p_coupon_code
    FOR UPDATE;

    IF NOT FOUND THEN
        RETURN jsonb_build_object(
            'success', false,
            'code', 'INVALID_COUPON'
        );
    END IF;

    -- 2. Validate coupon state
    IF v_coupon.status = 'expired' THEN
        RETURN jsonb_build_object('success', false, 'code', 'COUPON_EXPIRED');
    END IF;

    IF v_coupon.status = 'finished' THEN
        RETURN jsonb_build_object('success', false, 'code', 'COUPON_FULLY_REDEEMED');
    END IF;

    IF NOW() < v_coupon.starts_at THEN
        RETURN jsonb_build_object(
            'success', false,
            'code', 'COUPON_NOT_STARTED',
            'starts_at', v_coupon.starts_at
        );
    END IF;

    IF NOW() >= v_coupon.expires_at THEN
        RETURN jsonb_build_object('success', false, 'code', 'COUPON_EXPIRED');
    END IF;

    IF v_coupon.current_uses >= v_coupon.max_uses_total THEN
        RETURN jsonb_build_object('success', false, 'code', 'COUPON_FULLY_REDEEMED');
    END IF;

    -- 3. Prevent double‑claim by the same user
    PERFORM 1
    FROM coupon_history
    WHERE coupon_id = v_coupon.coupon_id
      AND user_telegram_id = p_telegram_id
    LIMIT 1;

    IF FOUND THEN
        RETURN jsonb_build_object('success', false, 'code', 'ALREADY_REDEEMED');
    END IF;

    -- 4. Atomically increment coupon usage and update status if needed
    UPDATE coupons
    SET
        current_uses = current_uses + 1,
        status = CASE
            WHEN current_uses + 1 >= max_uses_total THEN 'finished'
            ELSE status
        END
    WHERE coupon_id = v_coupon.coupon_id
      AND current_uses < max_uses_total;

    IF NOT FOUND THEN
        -- Race condition: another request used the last slot just before us
        RETURN jsonb_build_object('success', false, 'code', 'COUPON_FULLY_REDEEMED');
    END IF;

    -- 5. Credit the correct user wallet (defensively handle potential NULL)
    IF v_coupon.credit_wallet = 'withdrawal' THEN
        UPDATE users
        SET withdrawal_wallet = COALESCE(withdrawal_wallet, 0) + v_coupon.coupon_prize
        WHERE telegram_id = p_telegram_id;
    ELSE
        UPDATE users
        SET non_withdrawal_wallet = COALESCE(non_withdrawal_wallet, 0) + v_coupon.coupon_prize
        WHERE telegram_id = p_telegram_id;
    END IF;

    IF NOT FOUND THEN
        -- User does not exist (should not happen after previous checks)
        RETURN jsonb_build_object('success', false, 'code', 'USER_NOT_FOUND');
    END IF;

    -- 6. Record the claim in history
    INSERT INTO coupon_history (
        coupon_id,
        user_telegram_id,
        credited_amount,
        credit_wallet,
        claimed_at
    ) VALUES (
        v_coupon.coupon_id,
        p_telegram_id,
        v_coupon.coupon_prize,
        v_coupon.credit_wallet,
        NOW()
    )
    RETURNING claimed_id INTO v_claimed_id;

    -- 7. Success – return raw data only
    RETURN jsonb_build_object(
        'success', true,
        'coupon_prize', v_coupon.coupon_prize,
        'credited_amount', v_coupon.coupon_prize,
        'credit_wallet', v_coupon.credit_wallet,
        'coupon_code', p_coupon_code
    );

EXCEPTION
    WHEN unique_violation THEN
        RETURN jsonb_build_object('success', false, 'code', 'ALREADY_REDEEMED');
    WHEN OTHERS THEN
        RETURN jsonb_build_object('success', false, 'code', 'INTERNAL_ERROR');
END;
$$;

CREATE TABLE admin_credit_history (
     credit_id BIGSERIAL PRIMARY KEY, 
     amount NUMERIC(12,2),
     credited_wallet TEXT, 
     telegram_id BIGINT REFERENCES users(telegram_id) ON DELETE SET NULL,
     created_at         TIMESTAMPTZ DEFAULT NOW()

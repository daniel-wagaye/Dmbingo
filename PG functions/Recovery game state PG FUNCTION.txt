CREATE OR REPLACE FUNCTION recover_game_state()
RETURNS JSON
LANGUAGE plpgsql
AS $$
<<recovery>>
DECLARE
    current_game      RECORD;
    current_game_id   BIGINT;
    config            RECORD;
    winner            RECORD;
    global_status     TEXT;
    total_prize       NUMERIC(12,2);
    new_game_id       BIGINT;
    new_game_phase    TEXT;
    new_picking_ends  TIMESTAMPTZ;
    new_stake_amount  NUMERIC(12,2);
    winner_count      INT;
    has_winners       BOOLEAN;
    winner_list       JSONB := '[]'::JSONB;
    cents_total       BIGINT;
    base_cents        BIGINT;
    last_cents        BIGINT;
    share_cents       BIGINT;
    last_winner_id    BIGINT;
    result_json       JSONB;
BEGIN
    -- Advisory lock to serialise all recovery attempts.
    PERFORM pg_advisory_xact_lock(1);

    -- Lock the latest game row
    SELECT * INTO recovery.current_game
    FROM games
    ORDER BY game_id DESC
    LIMIT 1
    FOR UPDATE;

    -- If no game exists, create the first one
    IF NOT FOUND THEN
        INSERT INTO games (phase, stake_amount, minimum_player, picking_ends_at,
                           calling_started, shuffled_nums, prize_amount, house_profit, started_at)
        SELECT 'maintenance', stake_amount, minimum_player, NULL, FALSE, NULL, NULL, NULL, NOW()
        FROM game_config WHERE id = 1
        RETURNING game_id, phase INTO new_game_id, new_game_phase;

        RETURN json_build_object(
            'recovered', true,
            'action', 'initial_game_created',
            'new_game_id', new_game_id,
            'new_game_phase', new_game_phase
        );
    END IF;

    -- Store current game ID for SQL use
    current_game_id := recovery.current_game.game_id;

    -- Lock global status and config
    SELECT status INTO STRICT global_status
    FROM game_status
    WHERE id = 1
    FOR UPDATE;

    SELECT * INTO STRICT config
    FROM game_config
    WHERE id = 1
    FOR UPDATE;

    -- ====================================================================
    -- PHASE: maintenance
    -- ====================================================================
    IF recovery.current_game.phase = 'maintenance' THEN
        RETURN json_build_object(
            'recovered', true,
            'action', 'none',
            'game_id', recovery.current_game_id,
            'phase', recovery.current_game.phase
        );
    END IF;

    -- ====================================================================
    -- PHASE: picking
    -- ====================================================================
    IF recovery.current_game.phase = 'picking' THEN
        PERFORM 1 FROM player_picks
        WHERE player_picks.game_id = recovery.current_game_id
          AND player_picks.telegram_id IS NOT NULL
        FOR UPDATE;

        UPDATE users u
        SET withdrawal_wallet    = u.withdrawal_wallet + p.withdrawal_used,
            non_withdrawal_wallet = u.non_withdrawal_wallet + p.non_withdrawal_used
        FROM player_picks p
        WHERE p.game_id = recovery.current_game_id
          AND p.telegram_id IS NOT NULL
          AND p.telegram_id = u.telegram_id;

        UPDATE games
        SET phase = 'error',
            finished_at = NOW()
        WHERE game_id = recovery.current_game_id;

        -- Insert next game
        IF global_status = 'stopped' THEN
            INSERT INTO games (
                phase, stake_amount, minimum_player, picking_ends_at,
                calling_started, shuffled_nums, prize_amount, house_profit, started_at
            ) VALUES (
                'maintenance', 0, 0, NULL,
                FALSE, NULL, NULL, NULL, NOW()
            ) RETURNING game_id, phase INTO new_game_id, new_game_phase;
        ELSE
            INSERT INTO games (
                phase, stake_amount, minimum_player, picking_ends_at,
                calling_started, shuffled_nums, prize_amount, house_profit, started_at
            ) VALUES (
                'picking',
                config.stake_amount,
                config.minimum_player,
                NOW() + (config.picking_countdown_end_time * INTERVAL '1 second'),
                FALSE,
                NULL,
                NULL,
                NULL,
                NOW()
            ) RETURNING game_id, phase, picking_ends_at, stake_amount
            INTO new_game_id, new_game_phase, new_picking_ends, new_stake_amount;
        END IF;

        result_json := jsonb_build_object(
            'recovered', true,
            'action', 'refunded_picking',
            'previous_game_id', recovery.current_game_id,
            'new_game_id', new_game_id,
            'new_game_phase', new_game_phase
        );

        IF new_game_phase = 'picking' THEN
            result_json := result_json || jsonb_build_object(
                'picking_ends_at', new_picking_ends,
                'stake_amount', new_stake_amount
            );
        END IF;

        RETURN result_json;
    END IF;

    -- ====================================================================
    -- PHASE: started
    -- ====================================================================
    IF recovery.current_game.phase = 'started' THEN
        -- Fully qualify column names to avoid ambiguity
        SELECT EXISTS (
            SELECT 1 FROM player_picks
            WHERE player_picks.game_id = recovery.current_game_id
              AND player_picks.winner = TRUE
        ) INTO has_winners;

        IF has_winners THEN
            total_prize := COALESCE(recovery.current_game.prize_amount, 0);
            SELECT COUNT(*) INTO winner_count
            FROM player_picks
            WHERE player_picks.game_id = recovery.current_game_id
              AND player_picks.winner = TRUE;

            IF winner_count IS NULL OR winner_count = 0 THEN
                RAISE EXCEPTION 'Inconsistent state: has_winners = true but winner_count = 0 for game %', recovery.current_game_id;
            END IF;

            cents_total := (total_prize * 100)::BIGINT;
            base_cents := cents_total / winner_count;
            last_cents := cents_total - (base_cents * (winner_count - 1));

            SELECT player_picks.telegram_id INTO last_winner_id
            FROM player_picks
            WHERE player_picks.game_id = recovery.current_game_id
              AND player_picks.winner = TRUE
            ORDER BY player_picks.telegram_id DESC
            LIMIT 1;

            FOR winner IN
                SELECT player_picks.telegram_id, player_picks.board_id
                FROM player_picks
                WHERE player_picks.game_id = recovery.current_game_id
                  AND player_picks.winner = TRUE
                ORDER BY player_picks.telegram_id
                FOR UPDATE
            LOOP
                IF winner.telegram_id = last_winner_id THEN
                    share_cents := last_cents;
                ELSE
                    share_cents := base_cents;
                END IF;

                UPDATE users
                SET withdrawal_wallet = withdrawal_wallet + (share_cents::NUMERIC / 100)
                WHERE users.telegram_id = winner.telegram_id;

                INSERT INTO winners_history (game_id, telegram_id, board_id, credited_amount)
                VALUES (
                    recovery.current_game_id,
                    winner.telegram_id,
                    winner.board_id,
                    share_cents::NUMERIC / 100
                );

                winner_list := winner_list || jsonb_build_object(
                    'telegram_id', winner.telegram_id,
                    'board_id', winner.board_id,
                    'amount', share_cents::NUMERIC / 100
                );
            END LOOP;

            UPDATE games
            SET phase = 'error',
                finished_at = NOW()
            WHERE game_id = recovery.current_game_id;
        ELSE
            PERFORM 1 FROM player_picks
            WHERE player_picks.game_id = recovery.current_game_id
              AND player_picks.telegram_id IS NOT NULL
            FOR UPDATE;

            UPDATE users u
            SET withdrawal_wallet    = u.withdrawal_wallet + p.withdrawal_used,
                non_withdrawal_wallet = u.non_withdrawal_wallet + p.non_withdrawal_used
            FROM player_picks p
            WHERE p.game_id = recovery.current_game_id
              AND p.telegram_id IS NOT NULL
              AND p.telegram_id = u.telegram_id;

            UPDATE games
            SET phase = 'error',
                finished_at = NOW()
            WHERE game_id = recovery.current_game_id;
        END IF;

        -- Insert next game
        IF global_status = 'stopped' THEN
            INSERT INTO games (
                phase, stake_amount, minimum_player, picking_ends_at,
                calling_started, shuffled_nums, prize_amount, house_profit, started_at
            ) VALUES (
                'maintenance', 0, 0, NULL,
                FALSE, NULL, NULL, NULL, NOW()
            ) RETURNING game_id, phase INTO new_game_id, new_game_phase;
        ELSE
            INSERT INTO games (
                phase, stake_amount, minimum_player, picking_ends_at,
                calling_started, shuffled_nums, prize_amount, house_profit, started_at
            ) VALUES (
                'picking',
                config.stake_amount,
                config.minimum_player,
                NOW() + (config.picking_countdown_end_time * INTERVAL '1 second'),
                FALSE,
                NULL,
                NULL,
                NULL,
                NOW()
            ) RETURNING game_id, phase, picking_ends_at, stake_amount
            INTO new_game_id, new_game_phase, new_picking_ends, new_stake_amount;
        END IF;

        result_json := jsonb_build_object(
            'recovered', true,
            'action', CASE WHEN has_winners THEN 'credited_winners' ELSE 'refunded_started' END,
            'previous_game_id', recovery.current_game_id,
            'new_game_id', new_game_id,
            'new_game_phase', new_game_phase,
            'winners', winner_list
        );

        IF new_game_phase = 'picking' THEN
            result_json := result_json || jsonb_build_object(
                'picking_ends_at', new_picking_ends,
                'stake_amount', new_stake_amount
            );
        END IF;

        RETURN result_json;
    END IF;

    -- ====================================================================
    -- PHASE: winner_reveal or finished
    -- ====================================================================
    IF recovery.current_game.phase IN ('winner_reveal', 'finished') THEN
        UPDATE games
        SET phase = 'finished',
            finished_at = COALESCE(finished_at, NOW())
        WHERE game_id = recovery.current_game_id;

        IF global_status = 'stopped' THEN
            INSERT INTO games (
                phase, stake_amount, minimum_player, picking_ends_at,
                calling_started, shuffled_nums, prize_amount, house_profit, started_at
            ) VALUES (
                'maintenance', 0, 0, NULL,
                FALSE, NULL, NULL, NULL, NOW()
            ) RETURNING game_id, phase INTO new_game_id, new_game_phase;
        ELSE
            INSERT INTO games (
                phase, stake_amount, minimum_player, picking_ends_at,
                calling_started, shuffled_nums, prize_amount, house_profit, started_at
            ) VALUES (
                'picking',
                config.stake_amount,
                config.minimum_player,
                NOW() + (config.picking_countdown_end_time * INTERVAL '1 second'),
                FALSE,
                NULL,
                NULL,
                NULL,
                NOW()
            ) RETURNING game_id, phase, picking_ends_at, stake_amount
            INTO new_game_id, new_game_phase, new_picking_ends, new_stake_amount;
        END IF;

        result_json := jsonb_build_object(
            'recovered', true,
            'action', 'advanced_to_next_game',
            'previous_game_id', recovery.current_game_id,
            'new_game_id', new_game_id,
            'new_game_phase', new_game_phase
        );

        IF new_game_phase = 'picking' THEN
            result_json := result_json || jsonb_build_object(
                'picking_ends_at', new_picking_ends,
                'stake_amount', new_stake_amount
            );
        END IF;

        RETURN result_json;
    END IF;

    -- Unknown phase
    RETURN json_build_object(
        'recovered', true,
        'action', 'unknown_phase_ignored',
        'game_id', recovery.current_game_id,
        'phase', recovery.current_game.phase
    );
END;
$$;
